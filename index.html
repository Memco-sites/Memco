-- ===================================================================== -- MEMCO — Administration powers through the Permission Matrix -- -- Roles you create (Assistant Admin, Store Manager, …) can now be given -- admin powers by ticking them in the Permission Matrix: -- manageUsers Manage Users (add, edit, block users) -- managePermissions Permission Matrix and roles -- siteAccess Site Access screen -- -- Guard rails, always on: -- · only a Super Admin can make, change or delete a Super Admin -- · a Super Admin row can never be edited by anyone else -- -- Also redeploy the Edge Function "memco-users" with the new index.ts -- from this package (Verify JWT = OFF), otherwise creating accounts -- still needs a Super Admin. -- -- Run AFTER 1_memco_security.sql and 9_memco_site_access.sql. Safe to run again. -- ===================================================================== -- does my role have this tick in the Permission Matrix? create or replace function public.app_can(p_perm text) returns boolean language sql stable security definer set search_path = public as $$ select public.is_admin() or exists ( select 1 from public.profiles p join public.app_shared_state s on s.id = 'role-perms-matrix-v2' where p.id = auth.uid() and not p.blocked and jsonb_typeof(s.data) = 'object' and coalesce((s.data -> p.role ->> p_perm)::boolean, false) ); $$; revoke all on function public.app_can(text) from public, anon; grant execute on function public.app_can(text) to authenticated; -- --------------------------------------------------------------------- -- profiles: people with "Manage Users" may edit accounts — never a Super Admin -- --------------------------------------------------------------------- drop policy if exists profiles_admin_update on public.profiles; create policy profiles_admin_update on public.profiles for update to authenticated using (public.is_admin() or (public.app_can('manageUsers') and role <> 'Super Admin')) with check (public.is_admin() or (public.app_can('manageUsers') and role <> 'Super Admin')); -- --------------------------------------------------------------------- -- shared lists: users list and the permission matrix follow the ticks -- --------------------------------------------------------------------- drop policy if exists shared_insert on public.app_shared_state; drop policy if exists shared_update on public.app_shared_state; create policy shared_insert on public.app_shared_state for insert to authenticated with check ( public.is_active_user() and (public.is_admin() or (id = 'users-v1' and public.app_can('manageUsers')) or (id in ('role-perms-matrix-v2', 'custom-roles-v1') and public.app_can('managePermissions')) or id not in ('users-v1', 'role-perms-matrix-v2', 'custom-roles-v1')) ); create policy shared_update on public.app_shared_state for update to authenticated using ( public.is_active_user() and (public.is_admin() or (id = 'users-v1' and public.app_can('manageUsers')) or (id in ('role-perms-matrix-v2', 'custom-roles-v1') and public.app_can('managePermissions')) or id not in ('users-v1', 'role-perms-matrix-v2', 'custom-roles-v1')) ) with check ( public.is_active_user() and (public.is_admin() or (id = 'users-v1' and public.app_can('manageUsers')) or (id in ('role-perms-matrix-v2', 'custom-roles-v1') and public.app_can('managePermissions')) or id not in ('users-v1', 'role-perms-matrix-v2', 'custom-roles-v1')) ); -- --------------------------------------------------------------------- -- Site Access: any role with the "siteAccess" tick may use it -- (limited to his own site when his account has one) -- --------------------------------------------------------------------- create or replace function public.sa_scope() returns jsonb language plpgsql stable security definer set search_path = public as $$ declare p public.profiles; begin select * into p from public.profiles where id = auth.uid() and not blocked; if not found then return jsonb_build_object('can_grant', false); end if; if p.role in ('Super Admin', 'HR') then return jsonb_build_object('can_grant', true, 'all_sites', true, 'sites', '[]'::jsonb, 'office', true, 'role', p.role); end if; if p.role in ('Project Manager', 'Project Engineer', 'Site Engineer') or public.app_can('siteAccess') then return jsonb_build_object('can_grant', true, 'all_sites', coalesce(p.site, '') = '', 'sites', case when coalesce(p.site, '') = '' then '[]'::jsonb else jsonb_build_array(p.site) end, 'office', false, 'role', p.role); end if; return jsonb_build_object('can_grant', false, 'role', p.role); end $$;
